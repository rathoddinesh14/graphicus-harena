\begin{frame}{Performance \& Key Takeaways}
\begin{itemize}
    \item \textbf{Memory Footprint}: Approximately \textbf{200 MB} for a standard CT volume ($512 \times 512 \times 200$).
    \item \textbf{Complexity}:
    \begin{itemize}
        \item Parsing/Building: $O(N)$ voxels (One-time).
        \item Slice Extraction: $O(W \times H)$ (Real-time).
    \end{itemize}
    \item \textbf{Efficiency}: Leverages GPU hardware interpolation for smooth 2D/3D viewing.
\end{itemize}



\begin{block}{Advantage}
Web-based delivery allows for diagnostic-quality visualization on any device without software installation.
\end{block}
\end{frame}
