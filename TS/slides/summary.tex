\begin{alertblock}{The Transformation}
From \textbf{200 DICOM files} ($\approx$ 200MB) to \textbf{interactive 3D visualization} in the browser, all in real-time!
\end{alertblock}
\end{frame}

\begin{frame}{Performance \& Key Achievements}
\begin{itemize}
    \item \textbf{Memory Footprint}: Approximately \textbf{200 MB} for a standard CT volume ($512 \times 512 \times 200$).
    \item \textbf{Real-time Rendering}: 60 FPS interactive viewing with WebGL GPU acceleration.
    \item \textbf{Cross-platform}: Works on any device with a modern web browser.
    \item \textbf{Medical Accuracy}: Preserves Hounsfield Units and spatial relationships.
\end{itemize}

\vspace{0.5cm}

\textbf{Technical Highlights:}
\begin{itemize}
  \item \textbf{Data Pipeline}: DICOM parsing $\rightarrow$ Volume construction $\rightarrow$ Slice extraction
  \item \textbf{GPU Pipeline}: Texture upload $\rightarrow$ Shader processing $\rightarrow$ Canvas display
  \item \textbf{User Experience}: Orthogonal views with windowing controls
\end{itemize}

\begin{block}{Impact}
Web-based delivery allows for diagnostic-quality visualization on any device without software installation, democratizing access to advanced medical imaging tools.
\end{block}
\end{frame}
