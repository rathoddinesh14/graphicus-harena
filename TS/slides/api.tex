\section{API Integration: Bringing It All Together}

\begin{frame}{Core WebGL APIs: The Essential Toolkit}
\textbf{The Rendering Context}
\begin{itemize}
  \item \texttt{canvas.getContext('webgl')}: Gateway to the GPU.
  \item \texttt{gl.viewport()}: Defines the drawing "rectangle" in pixels.
\end{itemize}

\hrule % Visual separator
\vspace{0.5em}

\textbf{Shader Management Pipeline}
\begin{enumerate}
  \item \textbf{Create}: \texttt{gl.createShader(type)}
  \item \textbf{Source}: \texttt{gl.shaderSource()} (Injects GLSL string)
  \item \textbf{Compile}: \texttt{gl.compileShader()} $\to$ GPU machine code
  \item \textbf{Link}: \texttt{gl.linkProgram()} (Combines Vertex + Fragment)
\end{enumerate}

\begin{block}{Pro-Tip}
Always check \texttt{gl.getShaderParameter(..., gl.COMPILE\_STATUS)} to debug GLSL syntax errors!
\end{block}

\knowledgecheck{Error Handling}
What happens if a shader fails to compile? How would you debug this in a medical imaging application?
\end{frame}

\begin{frame}{Buffer \& Texture Operations: Data Management}
\textbf{Data Storage (VBOs)}
\begin{itemize}
  \item \texttt{STATIC\_DRAW}: Best for the static volume bounding box.
  \item \texttt{DYNAMIC\_DRAW}: Ideal for the 2D slice quad if it moves.
\end{itemize}

\hrule
\vspace{0.5em}

\textbf{Texture Mapping (The Heart of the Project)}
\begin{itemize}
  \item \texttt{gl.texImage2D()}: This is where our \textbf{Uint8Array} slice data gets uploaded.
  \item \textbf{Format}: \texttt{gl.LUMINANCE} (Saves memory for grayscale CT scans).
\end{itemize}

\begin{exampleblock}{Filtering}
Use \texttt{gl.LINEAR} for \texttt{MAG\_FILTER} to get smooth interpolation between pixels when zooming into a medical image.
\end{exampleblock}

\clinicalexample{Memory Optimization}
Using LUMINANCE format saves 75\% memory compared to RGBA. Critical for large medical volumes that must fit in GPU memory.
\end{frame}

\begin{frame}{Vertex Attributes \& Uniforms}
\textbf{The Bridge: \texttt{vertexAttribPointer}}
This command tells the GPU how to read our interleaved buffer:
\[ \text{Offset} = 0, \text{Stride} = 20 \text{ bytes} \]

\begin{itemize}
  \item \texttt{gl.enableVertexAttribArray()}: "Turns on" a shader variable.
\end{itemize}

\hrule
\vspace{0.5em}

\textbf{Global Constants (Uniforms)}
\begin{itemize}
  \item \texttt{gl.uniformMatrix4fv()}: Pass the Model-View-Projection (MVP) matrix.
  \item \texttt{gl.uniform1i()}: Tells the fragment shader which texture unit to sample.
\end{itemize}
\end{frame}

\vfill
\begin{alertblock}{DrawElements vs DrawArrays}
We use \texttt{drawElements} for meshes to reuse vertices via index buffers, reducing memory bandwidth.
\end{alertblock}
