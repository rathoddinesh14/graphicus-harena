\section{DICOM Parsing: Decoding Medical Data}

\begin{frame}
\frametitle{Chapter 2: Cracking the DICOM Code}

\begin{columns}[T]
\begin{column}{0.6\textwidth}
  \textbf{Dr. Chen's Challenge:} "I have 200 DICOM files from a chest CT. How do I read them?"
  
  \vspace{0.5cm}
  DICOM files are \textbf{binary} with \textbf{tagged structure}:
  \begin{itemize}
    \item \textbf{Group and Element tags}: (XXXX, XXXX) in hexadecimal
    \item \textbf{Value Representation (VR)}: Data type indicator
    \item \textbf{Length}: Bytes in the value field
    \item \textbf{Value}: The actual medical data
  \end{itemize}
\end{column}

A chest CT slice might be 512×512 pixels. The DICOM tag (0028,0010) stores the height (512), while (0028,0011) stores the width.

\begin{frame}[fragile]{DICOM Parsing: Extracting Metadata}
\begin{columns}
\begin{column}{0.5\textwidth}
\begin{lstlisting}[language=JavaScript]
// Extract Header Tags
const rows = dataSet.uint16('x00280010');
const cols = dataSet.uint16('x00280011');
const bits = dataSet.uint16('x00280100');
\end{lstlisting}
\end{column}
\begin{column}{0.5\textwidth}
\small
\textbf{Key Tags:}
\begin{itemize}
    \item \texttt{x00280010}: Image Height (Rows)
    \item \texttt{x00280011}: Image Width (Columns)
    \item \texttt{x00280100}: Bits Allocated (usually 16-bit for CT/MRI)
\end{itemize}
\end{column}
\end{columns}

\clinicalexample{CT Scan Parameters}
A typical chest CT: 512×512 pixels per slice, 16 bits per pixel, 200 slices total. This gives us 512×512×200 = 52.4 million voxels to process!
\end{frame}

\begin{frame}[fragile]{DICOM Parsing: Pixel Data Extraction}
\begin{lstlisting}[language=JavaScript]
// Locate the Pixel Data Element (7FE0,0010)
const pixelElement = dataSet.elements.x7fe00010;

// Create a view without copying the underlying buffer
const pixelData = new Uint16Array(
  byteArray.buffer, 
  pixelElement.dataOffset, 
  pixelElement.length / 2
);
\end{lstlisting}
\vfill
\begin{alertblock}{Memory Efficiency Critical}
Using \texttt{Uint16Array} on the existing \texttt{buffer} prevents a full memory copy, which is critical when loading 500+ slices in a web browser.
\end{alertblock}

\knowledgecheck{Data Types}
Why do we use Uint16Array for CT data? What would happen if we used Uint8Array instead?
\end{frame}

\begin{frame}[fragile]{DICOM Parsing: Spatial Positioning}
\begin{lstlisting}[language=JavaScript]
// Image Position (Patient) Tag
const posStr = dataSet.string('x00200032');
const [x, y, z] = posStr.split(' ').map(Number);
\end{lstlisting}
\vfill
\begin{itemize}
    \item \textbf{Image Position (Patient):} Specifies the $x, y, z$ coordinates of the upper left hand corner of the slice.
    \item \textbf{Stacking:} We use the \texttt{z} value to sort slices into a coherent 3D volume.
\end{itemize}

\clinicalexample{Slice Spacing}
In a chest CT, slices might be positioned at z = -150.0mm, -148.5mm, -147.0mm, etc. The spacing between slices (1.5mm) determines the resolution in the cranial-caudal direction.
\end{frame}

\begin{frame}
\frametitle{Data Type Handling: From Bits to Meaningful Values}

\begin{columns}[T]
\begin{column}{0.5\textwidth}
  \begin{itemize}
    \item \textbf{8-bit images}: Uint8Array (common in radiography)
    \item \textbf{16-bit images}: Uint16Array (common in CT/MRI)
    \item \textbf{Byte ordering}: Little-endian on most systems
    \item \textbf{Pixel representation}: Usually unsigned (0 to max)
    \item \textbf{Hounsfield Units} in CT: Requires proper windowing
  \end{itemize}
\end{column}

\clinicalexample{Hounsfield Units}
Water = 0 HU, Air = -1000 HU, Bone = +1000 HU. This standardized scale allows radiologists to identify tissues regardless of the scanner used.

\begin{frame}
\frametitle{Slice Collection: Building the Volume}

\textbf{Slice Collection:}
\begin{itemize}
  \item Multiple DICOM files $\rightarrow$ one file per slice
  \item Files must be sorted by spatial position
  \item Use \texttt{ImagePositionPatient} (most reliable)
  \item Fallback to \texttt{InstanceNumber}
\end{itemize}

\vspace{0.5cm}

\begin{alertblock}{Critical Sorting Issue}
If slices aren't sorted correctly, the 3D volume will have anatomical structures in wrong positions, leading to misdiagnosis!
\end{alertblock}
