\section{Interactive Slice Rendering}

\begin{frame}[allowframebreaks]
\frametitle{Interactive Slice Rendering}
\textbf{The Goal:} Render a single 2D slice as a "planar cross-section" in 3D space.

\begin{block}{The Reconstruction Strategy}
  \begin{enumerate}
    \item \textbf{Sample:} Extract a 1D buffer from the 3D volume.
    \item \textbf{Upload:} Bind as a 2D Texture via \texttt{gl.texImage2D}.
    \item \textbf{Project:} Draw a 2-triangle "Quad" positioned at the slice index.
  \end{enumerate}
\end{block}
\end{frame}

\begin{frame}[fragile]
\frametitle{Slice Texture Update}
\begin{lstlisting}
// Bind the texture unit
this.gl.bindTexture(this.gl.TEXTURE_2D, this.sliceTexture);

// Upload slice data (Grayscale/LUMINANCE)
this.gl.texImage2D(
  this.gl.TEXTURE_2D, 0, this.gl.LUMINANCE,
  slice.width, slice.height, 0,
  this.gl.LUMINANCE, this.gl.UNSIGNED_BYTE,
  slice.data
);

// Bilinear Interpolation for smooth zooming
this.gl.texParameteri(this.gl.TEXTURE_2D, 
  this.gl.TEXTURE_MIN_FILTER, this.gl.LINEAR);
\end{lstlisting}

\small \textbf{Why \texttt{LUMINANCE}?} DICOM data is single-channel. Using \texttt{LUMINANCE} uses 1 byte per pixel, whereas \texttt{RGBA} would use 4 bytes, saving 75\% GPU memory.
\end{frame}

\begin{exampleblock}{Coordinate Normalization}
We scale the volume from its raw pixel units (e.g., $512 \times 512 \times 200$) to a unit cube $[0, 1]^3$. This makes camera math and rotations much simpler.
\end{exampleblock}
