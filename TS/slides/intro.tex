\begin{frame}
\frametitle{What is DICOM? The Language of Medical Imaging}

\begin{columns}[T]
\begin{column}{0.5\textwidth}
  \begin{itemize}
    \item \textbf{DICOM} = Digital Imaging and Communications in Medicine
    \item International standard for medical imaging data
    \item Contains both \textbf{metadata} and \textbf{pixel data}
    \item Common in CT, MRI, X-ray, and ultrasound
    \item \textbf{One file per slice} forms a 3D volume
  \end{itemize}
\end{column}
\begin{column}{0.45\textwidth}
  \begin{center}
    \begin{tikzpicture}[scale=0.7]

\clinicalexample{Lung Cancer Screening}
A single chest CT scan might contain 200+ DICOM files, each representing a 1-2mm thick slice through the patient's thorax. The radiologist needs to mentally reconstruct these into a 3D understanding of the anatomy.
\end{frame}

\begin{frame}
\frametitle{Our Solution: Graphicus-Harena}

\begin{columns}[T]
\begin{column}{0.6\textwidth}
  \begin{enumerate}
    \item \textbf{Parse} multiple DICOM files into a coherent 3D volume
    \item \textbf{Extract} orthogonal slices (axial, sagittal, coronal) on demand
    \item \textbf{Render} using WebGL for GPU-accelerated visualization
    \item \textbf{Interact} with mouse controls for real-time exploration
  \end{enumerate}
\end{column}
